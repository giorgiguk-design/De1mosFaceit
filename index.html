async function loadQuestsProgress() { if (!currentUser) return { daily: [], weekly: [] }; const dailyReset = getDailyResetTime(); const weeklyReset = getWeeklyResetTime(); try { const { data } = await sb.from('quests_progress') .select('*') .eq('telegram_id', currentUser.telegram_id) .gte('reset_at', new Date(Date.now() - 24 * 60 * 60 * 1000).toISOString()); const sameTime = (a, b) => { try { return Math.abs(new Date(a).getTime() - new Date(b).getTime()) < 1000; } catch (e) { return false; } }; const daily = QUESTS_CONFIG.daily.map(q => { const found = (data || []).find(r => r.quest_id === q.id && sameTime(r.reset_at, dailyReset) ); return { ...q, progress: found?.progress || 0, claimed: found?.claimed || false, reset_at: dailyReset, type: 'daily', }; }); const weekly = QUESTS_CONFIG.weekly.map(q => { const found = (data || []).find(r => r.quest_id === q.id && sameTime(r.reset_at, weeklyReset) ); return { ...q, progress: found?.progress || 0, claimed: found?.claimed || false, reset_at: weeklyReset, type: 'weekly', }; }); // ⬇️ ВОТ ЭТОГО НЕ ХВАТАЛО ⬇️ return { daily, weekly }; } catch (e) { console.error('loadQuestsProgress:', e); return { daily: [], weekly: [] }; } }
